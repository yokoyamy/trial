function shape(ctx,e){
 ctx.beginPath();
 if(e.shape==='ellipse')ctx.ellipse(e.x+e.w/2,e.y+e.h/2,e.w/2,e.h/2,0,0,Math.PI*2);

}
function port(e,p){
 if(e.shape==='ellipse')return{x:e.x+e.w/2+Math.cos(p.angle||0)*e.w/2,y:e.y+e.h/2+Math.sin(p.angle||0)*e.h/2};

 return p.side==='left'?{x:e.x,y:e.y+e.h*r}:p.side==='right'?{x:e.x+e.w,y:e.y+e.h*r}:p.side==='top'?{x:e.x+e.w*r,y:e.y}:{x:e.x+e.w*r,y:e.y+e.h};
}
function nearest(e,q){
 if(e.shape==='ellipse')return{angle:Math.atan2((q.y-e.y-e.h/2)/(e.h/2),(q.x-e.x-e.w/2)/(e.w/2))};
 return[
  {side:'left',r:clamp((q.y-e.y)/e.h,.05,.95)},
  {side:'right',r:clamp((q.y-e.y)/e.h,.05,.95)},
  {side:'top',r:clamp((q.x-e.x)/e.w,.05,.95)},
  {side:'bottom',r:clamp((q.x-e.x)/e.w,.05,.95)}
 ].sort((a,b)=>{
  const p=port(e,a),r=port(e,b);
  return Math.hypot(p.x-q.x,p.y-q.y)-Math.hypot(r.x-q.x,r.y-q.y);
 })[0];
}

}

async function render(){
 if(!canvas.captureStream)throw Error('このブラウザは編集動画の生成に対応していません');
 if(skip(0)>=duration())throw Error('全区間がスキップされているため保存できません');
 const time=video.currentTime,muted=video.muted,rate=video.playbackRate;
 const output=document.createElement('canvas');output.width=canvas.width;output.height=canvas.height;
 const ctx=output.getContext('2d');
 let stream,recorder,frameID,detached=false;
 video.pause();
 try{
  await audioReady();audioSource.disconnect(audio.destination);detached=true;
  await seek(skip(0));
  const draw=()=>{ctx.drawImage(video,0,0,output.width,output.height);paint(ctx,video.currentTime,true)};
  draw();stream=output.captureStream(30);
  audioSink.stream.getAudioTracks().forEach(t=>stream.addTrack(t.clone()));
  recorder=createRecorder(stream);const result=collect(recorder);result.catch(()=>{});
  const finished=new Promise((resolve,reject)=>{
   const loop=async()=>{
    try{
     if(video.ended||video.currentTime>=duration()){resolve();return}
     const next=skip(video.currentTime);
     if(next>video.currentTime){
      video.pause();await seek(next);
      if(next>=duration()){resolve();return}
      await video.play();
     }
     if(video.readyState>=2&&!video.seeking)draw();
     frameID=requestAnimationFrame(loop);
    }catch(error){reject(error)}
   };
   frameID=requestAnimationFrame(loop);
  });
  finished.catch(()=>{});
  recorder.start(250);video.muted=false;video.playbackRate=1;
  await video.play();
  await Promise.race([finished,result.then(()=>{throw Error('書き出しが途中で終了しました')})]);
  video.pause();recorder.stop();return await result;
 }finally{
  cancelAnimationFrame(frameID);video.pause();
  if(recorder?.state==='recording')recorder.stop();
  stream?.getTracks().forEach(t=>t.stop());
  if(detached)audioSource.connect(audio.destination);
  video.muted=muted;video.playbackRate=rate;await seek(Math.min(time,duration()));
 }
}

 if(!busy&&editing&&!video.paused&&!skipping){
  const next=skip(video.currentTime);
  if(next>video.currentTime){
   skipping=true;
   try{await seek(next);if(next<duration())await video.play()}
   catch(error){message(error.message)}
   finally{skipping=false}
  }
 }
 paint();
 if(editing){
  const start=range().start;
  if(Math.abs(start-lastStart)>.05){lastStart=start;tracks()}
  else clock();
 }
 requestAnimationFrame(loop);
}
try{
 db=await new Promise((resolve,reject)=>{
  const request=indexedDB.open('VideoEditorClean',1);
  request.onupgradeneeded=()=>request.result.createObjectStore('data');
  request.onsuccess=()=>resolve(request.result);request.onerror=()=>reject(request.error);
 });
 model=await read('model')||{folders:[makeFolder('デフォルトフォルダ')]};
 tree();update();requestAnimationFrame(loop);
}catch(error){
 message('保存領域を初期化できませんでした：'+error.message);
 document.querySelectorAll('button').forEach(button=>button.disabled=true);
}
})();

<?php
declare(strict_types=1);

/*
 * PHP 8.1以上 / Apache 2.4 / HTTPSまたはlocalhost
 * storageへのHTTPアクセスにはApacheのAllowOverride有効化が必要。
 * 既存データは初期化しない。
 */

const ROOT = __DIR__ . '/storage';
const SCHEMA = 1;

final class AppError extends RuntimeException {
    public function __construct(
        public readonly string $kind,
        string $message,
        int $status = 400
    ) {
        parent::__construct($message, $status);
    }
}

function fail(string $kind, string $message, int $status = 400): never {
    throw new AppError($kind, $message, $status);
}

function uuid(): string {
    $b = random_bytes(16);
    $b[6] = chr((ord($b[6]) & 15) | 64);
    $b[8] = chr((ord($b[8]) & 63) | 128);
    $s = bin2hex($b);
    return substr($s,0,8).'-'.substr($s,8,4).'-'.substr($s,12,4).
        '-'.substr($s,16,4).'-'.substr($s,20);
}

function now(): string {
    return gmdate('Y-m-d\TH:i:s\Z');
}

function jsonText(mixed $value): string {
    return json_encode(
        $value,
        JSON_UNESCAPED_UNICODE |
        JSON_UNESCAPED_SLASHES |
        JSON_THROW_ON_ERROR
    );
}

function readJson(string $name): array {
    $text = file_get_contents(ROOT.'/'.$name);
    if ($text === false) fail('STORAGE_ERROR','管理データを読めません。',500);
    try {
        $data = json_decode($text,true,512,JSON_THROW_ON_ERROR);
    } catch (JsonException) {
        fail('STORAGE_ERROR','管理データが破損しています。',500);
    }
    if (!is_array($data)) fail('STORAGE_ERROR','管理データが不正です。',500);
    return $data;
}

function atomicWrite(string $name, array $data): void {
    $tmp = tempnam(ROOT.'/tmp','json-');
    if ($tmp === false) fail('SAVE_FAILED','一時ファイルを作成できません。',500);
    $fp = null;
    try {
        $text = jsonText($data);
        $fp = fopen($tmp,'wb');
        if (!$fp) fail('SAVE_FAILED','管理データを書き込めません。',500);
        $offset = 0;
        while ($offset < strlen($text)) {
            $count = fwrite($fp,substr($text,$offset));
            if ($count === false || $count === 0)
                fail('SAVE_FAILED','管理データの書き込みに失敗しました。',500);
            $offset += $count;
        }
        if (!fflush($fp)) fail('SAVE_FAILED','書き込みを確定できません。',500);
        if (function_exists('fsync') && !fsync($fp))
            fail('SAVE_FAILED','書き込みを同期できません。',500);
        fclose($fp);
        $fp = null;
        if (!rename($tmp,ROOT.'/'.$name))
            fail('SAVE_FAILED','管理データを置き換えられません。',500);
    } finally {
        if (is_resource($fp)) fclose($fp);
        if (is_file($tmp)) @unlink($tmp);
    }
}

function id(mixed $value): string {
    if (!is_string($value) ||
        !preg_match('/^[0-9a-f]{8}-(?:[0-9a-f]{4}-){3}[0-9a-f]{12}$/i',$value))
        fail('INVALID_INPUT','UUIDが不正です。');
    return $value;
}

function nameValue(mixed $value): string {
    if (!is_string($value) || trim($value)==='' || strlen($value)>600)
        fail('INVALID_INPUT','名前が不正です。');
    return trim($value);
}

function numberValue(mixed $value, string $label): float {
    if ((!is_int($value) && !is_float($value)) || !is_finite((float)$value))
        fail('INVALID_INPUT',$label.'が不正です。');
    return (float)$value;
}

function indexOf(array $rows, string $value): int {
    foreach ($rows as $i=>$row)
        if (($row['id']??null)===$value) return $i;
    fail('NOT_FOUND','対象がありません。',404);
}

function filePath(array $file): string {
    $relative = $file['storagePath']??null;
    if (!is_string($relative) ||
        !preg_match('~^files/[0-9a-f-]{36}\.(mp4|webm|mov)$~i',$relative))
        fail('STORAGE_ERROR','保存ファイル情報が不正です。',500);
    return ROOT.'/'.$relative;
}

function canonical(mixed $value): mixed {
    if (!is_array($value)) return $value;
    if (!array_is_list($value)) ksort($value);
    foreach ($value as $key=>$item) $value[$key]=canonical($item);
    return $value;
}

function validateEdit(mixed $data): array {
    if (!is_array($data) ||
        ($data['schemaVersion']??null)!==SCHEMA ||
        ($data['coordinateSystem']??null)!=='source-pixels' ||
        !is_array($data['elements']??null) ||
        !array_is_list($data['elements']))
        fail('INVALID_INPUT','編集データ形式が不正です。');

    $sw=numberValue($data['sourceWidth']??null,'元動画幅');
    $sh=numberValue($data['sourceHeight']??null,'元動画高さ');
    if ($sw<=0 || $sh<=0) fail('INVALID_INPUT','元動画サイズが不正です。');

    $elements=[];
    $color=function(mixed $v): void {
        if (!is_string($v) ||
            !preg_match('/^(#[0-9a-f]{6}|transparent)$/i',$v))
            fail('INVALID_INPUT','色が不正です。');
    };

    foreach ($data['elements'] as $e) {
        if (!is_array($e)) fail('INVALID_INPUT','編集要素が不正です。');
        $eid=id($e['id']??null);
        if (isset($elements[$eid])) fail('INVALID_INPUT','要素IDが重複しています。');
        $type=$e['type']??null;
        if (!in_array($type,['A','B','C','D','E','F'],true))
            fail('INVALID_INPUT','要素種類が不正です。');

        if ($type==='F') {
            if (numberValue($e['time']??null,'PDF位置')<0)
                fail('INVALID_INPUT','PDF位置が不正です。');
        } elseif ($type!=='E') {
            $start=numberValue($e['start']??null,'開始時刻');
            $end=numberValue($e['end']??null,'終了時刻');
            if ($start<0 || $end<=$start)
                fail('INVALID_INPUT','表示区間が不正です。');
        }

        if (in_array($type,['A','B','C'],true)) {
            foreach (['x','y','w','h','width'] as $key)
                if (numberValue($e[$key]??null,$key)<0)
                    fail('INVALID_INPUT','座標またはサイズが不正です。');
            if ($e['w']<=0 || $e['h']<=0 ||
                $e['x']+$e['w']>$sw+0.01 ||
                $e['y']+$e['h']>$sh+0.01 ||
                !in_array($e['shape']??null,['rect','ellipse','round'],true))
                fail('INVALID_INPUT','図形範囲が不正です。');
            $color($e['background']??null);
        }

        if (in_array($type,['A','B','C','E'],true)) {
            $color($e['stroke']??null);
            if (numberValue($e['width']??null,'線幅')<0 ||
                !in_array($e['dash']??null,['solid','dash','dot','chain'],true))
                fail('INVALID_INPUT','線書式が不正です。');
        }

        if ($type==='A') {
            $size=numberValue($e['size']??null,'文字サイズ');
            if (!is_string($e['text']??null) || $size<10 || $size>40 ||
                !in_array($e['font']??null,['gothic','mincho'],true))
                fail('INVALID_INPUT','コメント書式が不正です。');
            $color($e['color']??null);
        }
        if ($type==='C' && numberValue($e['zoom']??null,'拡大倍率')<1)
            fail('INVALID_INPUT','拡大倍率は1以上にしてください。');

        if ($type==='E') {
            id($e['ownerId']??null);
            id($e['commentId']??null);
            if (!in_array($e['line']??null,['straight','orthogonal','wave'],true) ||
                !in_array($e['terminal']??null,['none','arrow','circle'],true))
                fail('INVALID_INPUT','接続線書式が不正です。');
        }
        $elements[$eid]=$e;
    }

    foreach ($elements as $e) {
        if ($e['type']!=='E') continue;
        $owner=$elements[$e['ownerId']]??null;
        $comment=$elements[$e['commentId']]??null;
        if (!$owner || !$comment ||
            !in_array($owner['type'],['B','C'],true) || $comment['type']!=='A')
            fail('INVALID_INPUT','接続先が不正です。');
        foreach (['from'=>$owner,'to'=>$comment] as $key=>$target) {
            $position=$e[$key]??null;
            if (!is_array($position)) fail('INVALID_INPUT','接続位置が不正です。');
            if ($target['shape']==='ellipse') {
                numberValue($position['angle']??null,'接続角度');
            } else {
                $ratio=numberValue($position['ratio']??null,'接続比率');
                if ($ratio<0 || $ratio>1 ||
                    !in_array($position['side']??null,['left','right','top','bottom'],true))
                    fail('INVALID_INPUT','接続位置が不正です。');
            }
        }
    }
    return $data;
}

function receiveFile(): array {
    $upload=$_FILES['file']??null;
    if (!$upload || ($upload['error']??-1)!==UPLOAD_ERR_OK ||
        !is_uploaded_file($upload['tmp_name']))
        fail('UPLOAD_FAILED','動画を受信できません。');
    if (!class_exists('finfo'))
        fail('SERVER_CONFIGURATION','fileinfo拡張が必要です。',500);

    $mime=(new finfo(FILEINFO_MIME_TYPE))->file($upload['tmp_name']);
    $ext=match($mime) {
        'video/mp4'=>'mp4',
        'video/webm'=>'webm',
        'video/quicktime'=>'mov',
        default=>null
    };
    if (!$ext) fail('INVALID_INPUT','MP4、WebM、MOVを指定してください。');

    $fid=uuid();
    $temp=ROOT.'/tmp/'.$fid;
    if (!move_uploaded_file($upload['tmp_name'],$temp))
        fail('SAVE_FAILED','動画を一時保存できません。',500);
    $size=filesize($temp);
    if ($size===false || $size<=0) {
        @unlink($temp);
        fail('INVALID_INPUT','動画ファイルが空です。');
    }
    return [
        'file'=>[
            'id'=>$fid,'storagePath'=>"files/$fid.$ext",
            'mimeType'=>$mime,'size'=>$size,'createdAt'=>now()
        ],
        'temp'=>$temp
    ];
}

function recover(): void {
    if (!is_file(ROOT.'/transaction.json')) return;
    $tx=readJson('transaction.json');
    atomicWrite('metadata.json',$tx['metadata']);
    atomicWrite('requests.json',$tx['requests']);
    atomicWrite('deletion-retry.json',$tx['deletions']);
    if (!unlink(ROOT.'/transaction.json'))
        fail('STORAGE_ERROR','復旧記録を確定できません。',500);
}

function collectFiles(array $metadata): void {
    $referenced=[];
    foreach ($metadata['files'] as $file) $referenced[filePath($file)]=true;
    $pending=[];
    foreach (readJson('deletion-retry.json') as $file) {
        $path=filePath($file);
        if (!isset($referenced[$path]) && is_file($path) && !@unlink($path))
            $pending[$file['id']]=$file;
    }
    foreach (glob(ROOT.'/files/*')?:[] as $path) {
        if (!is_file($path) || isset($referenced[$path])) continue;
        if (!@unlink($path)) {
            $base=basename($path);
            if (preg_match('/^([0-9a-f-]{36})\.(mp4|webm|mov)$/i',$base,$m)) {
                $pending[$m[1]]=[
                    'id'=>$m[1],'storagePath'=>'files/'.$base,
                    'mimeType'=>'application/octet-stream',
                    'size'=>filesize($path)?:0,'createdAt'=>now()
                ];
            }
        }
    }
    foreach (glob(ROOT.'/tmp/*')?:[] as $path)
        if (is_file($path)) @unlink($path);
    atomicWrite('deletion-retry.json',array_values($pending));
}

function answer(array $data, ?string $requestId=null): never {
    header('Content-Type: application/json; charset=utf-8');
    header('Cache-Control: no-store');
    echo jsonText(['ok'=>true,'requestId'=>$requestId,'data'=>$data]);
    exit;
}

$lock=null;
$requestId=null;

try {
    foreach ([ROOT,ROOT.'/files',ROOT.'/tmp'] as $directory) {
        if (!is_dir($directory) &&
            !mkdir($directory,0700,true) && !is_dir($directory))
            fail('STORAGE_ERROR','保存領域を作成できません。',500);
        if (!is_readable($directory) || !is_writable($directory))
            fail('STORAGE_ERROR','保存領域の読み書き権限が必要です。',500);
    }

    $deny="Require all denied\n";
    if (!is_file(ROOT.'/.htaccess') &&
        file_put_contents(ROOT.'/.htaccess',$deny)!==strlen($deny))
        fail('STORAGE_ERROR','アクセス制御を作成できません。',500);
    if (trim((string)file_get_contents(ROOT.'/.htaccess'))!==trim($deny))
        fail('STORAGE_ERROR','storageのアクセス制御を確認してください。',500);

    $lock=fopen(ROOT.'/.lock','c');
    if (!$lock || !flock($lock,LOCK_EX))
        fail('STORAGE_ERROR','排他制御に失敗しました。',500);

    if (!is_file(ROOT.'/metadata.json')) {
        $time=now();
        atomicWrite('metadata.json',[
            'schemaVersion'=>SCHEMA,
            'folders'=>[[
                'id'=>uuid(),'parentFolderId'=>null,'name'=>'デフォルト',
                'createdAt'=>$time,'updatedAt'=>$time
            ]],
            'videos'=>[],'edits'=>[],'files'=>[]
        ]);
    }
    foreach (['requests.json','deletion-retry.json'] as $filename)
        if (!is_file(ROOT.'/'.$filename)) atomicWrite($filename,[]);

    $metadata=readJson('metadata.json');
    if (($metadata['schemaVersion']??null)!==SCHEMA)
        fail('SCHEMA_UNSUPPORTED','未対応のデータ形式です。上書きしません。',409);

    recover();
    $metadata=readJson('metadata.json');
    foreach (['folders','videos','edits','files'] as $key)
        if (!isset($metadata[$key]) || !is_array($metadata[$key]) ||
            !array_is_list($metadata[$key]))
            fail('STORAGE_ERROR','管理データ構造が不正です。',500);

    $method=$_SERVER['REQUEST_METHOD']??'GET';

    if (!isset($_GET['api'])) {
        collectFiles($metadata);
        flock($lock,LOCK_UN);
        fclose($lock);
        $lock=null;
    } else {
        if ($_GET['api']!=='v1') fail('NOT_FOUND','APIがありません。',404);
        $action=(string)($_GET['action']??'');

        if ($method==='GET') {
            if ($action==='list') {
                $public=$metadata;
                foreach ($public['files'] as &$file) unset($file['storagePath']);
                unset($file);
                answer($public);
            }

            if ($action==='edit.get') {
                answer($metadata['edits'][indexOf(
                    $metadata['edits'],id($_GET['id']??null)
                )]);
            }

            if (!in_array($action,['file.stream','file.download'],true))
                fail('NOT_FOUND','操作がありません。',404);

            $file=$metadata['files'][indexOf(
                $metadata['files'],id($_GET['id']??null)
            )];
            $path=filePath($file);
            $fp=@fopen($path,'rb');
            if (!$fp) fail('NOT_FOUND','動画ファイルがありません。',404);
            $size=filesize($path);
            if (!$size) fail('STORAGE_ERROR','動画ファイルが空です。',500);
            $start=0;
            $end=$size-1;

            if (isset($_SERVER['HTTP_RANGE'])) {
                $range=$_SERVER['HTTP_RANGE'];
                if (!preg_match('/^bytes=(\d*)-(\d*)$/',$range,$parts) ||
                    ($parts[1]==='' && $parts[2]==='')) {
                    header("Content-Range: bytes */$size");
                    fail('INVALID_RANGE','Rangeが不正です。',416);
                }
                if ($parts[1]==='') {
                    $suffix=(int)$parts[2];
                    if ($suffix<=0) {
                        header("Content-Range: bytes */$size");
                        fail('INVALID_RANGE','Rangeが不正です。',416);
                    }
                    $start=max(0,$size-$suffix);
                } else {
                    $start=(int)$parts[1];
                    if ($parts[2]!=='') $end=min($end,(int)$parts[2]);
                }
                if ($start>$end || $start>=$size) {
                    header("Content-Range: bytes */$size");
                    fail('INVALID_RANGE','Rangeが範囲外です。',416);
                }
                http_response_code(206);
                header("Content-Range: bytes $start-$end/$size");
            }

            header('Content-Type: '.$file['mimeType']);
            header('X-Content-Type-Options: nosniff');
            header('Accept-Ranges: bytes');
            header('Content-Length: '.($end-$start+1));
            if ($action==='file.download') {
                $display='動画';
                foreach ($metadata['videos'] as $video)
                    if ($video['originalFileId']===$file['id']) $display=$video['name'];
                foreach ($metadata['edits'] as $edit)
                    if ($edit['renderedFileId']===$file['id']) $display=$edit['name'];
                $extension=pathinfo($path,PATHINFO_EXTENSION);
                header("Content-Disposition: attachment; filename=\"video.$extension\"; ".
                    "filename*=UTF-8''".rawurlencode($display.'.'.$extension));
            }
            flock($lock,LOCK_UN);
            fclose($lock);
            $lock=null;
            fseek($fp,$start);
            $remaining=$end-$start+1;
            while ($remaining>0 && !connection_aborted()) {
                $chunk=fread($fp,min(1048576,$remaining));
                if ($chunk===false || $chunk==='') break;
                echo $chunk;
                $remaining-=strlen($chunk);
            }
            fclose($fp);
            exit;
        }

        if ($method!=='POST') fail('METHOD_NOT_ALLOWED','POSTを使用してください。',405);

        try {
            $request=json_decode(
                (string)($_POST['metadata']??file_get_contents('php://input')),
                true,512,JSON_THROW_ON_ERROR
            );
        } catch (JsonException) {
            fail('INVALID_INPUT','要求JSONが不正です。');
        }

        $requestId=id($request['requestId']??null);
        $payload=$request['payload']??null;
        if (!is_array($payload)) fail('INVALID_INPUT','要求内容が不正です。');

        $uploadHash=null;
        if (isset($_FILES['file']) &&
            ($_FILES['file']['error']??-1)===UPLOAD_ERR_OK &&
            is_uploaded_file($_FILES['file']['tmp_name']))
            $uploadHash=hash_file('sha256',$_FILES['file']['tmp_name']);

        $hash=hash('sha256',jsonText(canonical([$action,$payload,$uploadHash])));
        $requests=readJson('requests.json');
        if (isset($requests[$requestId])) {
            if ($requests[$requestId]['hash']!==$hash)
                fail('REQUEST_CONFLICT','同じ要求IDの内容が異なります。',409);
            answer($requests[$requestId]['data'],$requestId);
        }

        collectFiles($metadata);
        $entityId=id($payload['id']??null);
        $time=now();
        $result=['id'=>$entityId];
        $removed=[];
        $incoming=null;

        try {
            if ($action==='folder.create') {
                foreach ($metadata['folders'] as $folder)
                    if ($folder['id']===$entityId)
                        fail('CONFLICT','フォルダIDが存在します。',409);
                $parent=$payload['parentFolderId']??null;
                if ($parent!==null) indexOf($metadata['folders'],id($parent));
                $metadata['folders'][]=[
                    'id'=>$entityId,'parentFolderId'=>$parent,
                    'name'=>nameValue($payload['name']??null),
                    'createdAt'=>$time,'updatedAt'=>$time
                ];
            } elseif (in_array($action,[
                'folder.rename','video.rename','edit.rename'
            ],true)) {
                $key=explode('.',$action)[0].'s';
                $i=indexOf($metadata[$key],$entityId);
                $metadata[$key][$i]['name']=nameValue($payload['name']??null);
                $metadata[$key][$i]['updatedAt']=$time;
            } elseif ($action==='video.move') {
                $i=indexOf($metadata['videos'],$entityId);
                $folderId=id($payload['folderId']??null);
                indexOf($metadata['folders'],$folderId);
                $metadata['videos'][$i]['folderId']=$folderId;
                $metadata['videos'][$i]['updatedAt']=$time;
            } elseif ($action==='video.upload') {
                foreach ($metadata['videos'] as $video)
                    if ($video['id']===$entityId)
                        fail('CONFLICT','動画IDが存在します。',409);
                $folderId=id($payload['folderId']??null);
                indexOf($metadata['folders'],$folderId);
                $display=nameValue($payload['name']??null);
                $incoming=receiveFile();
                $metadata['files'][]=$incoming['file'];
                $metadata['videos'][]=[
                    'id'=>$entityId,'folderId'=>$folderId,'name'=>$display,
                    'originalFileId'=>$incoming['file']['id'],
                    'createdAt'=>$time,'updatedAt'=>$time
                ];
            } elseif ($action==='edit.save') {
                $videoId=id($payload['videoId']??null);
                indexOf($metadata['videos'],$videoId);
                $data=validateEdit($payload['editData']??null);
                $display=nameValue($payload['name']??null);
                $i=null;
                foreach ($metadata['edits'] as $k=>$edit)
                    if ($edit['id']===$entityId) $i=$k;
                $version=$i===null?0:$metadata['edits'][$i]['version'];
                if (($payload['expectedVersion']??null)!==$version ||
                    ($i!==null && $metadata['edits'][$i]['videoId']!==$videoId))
                    fail('VERSION_CONFLICT','編集版が競合しました。再取得してください。',409);

                $incoming=receiveFile();
                $metadata['files'][]=$incoming['file'];
                $result=[
                    'id'=>$entityId,'videoId'=>$videoId,'name'=>$display,
                    'version'=>$version+1,'editData'=>$data,
                    'renderedFileId'=>$incoming['file']['id'],
                    'createdAt'=>$i===null?$time:$metadata['edits'][$i]['createdAt'],
                    'updatedAt'=>$time
                ];
                if ($i===null) $metadata['edits'][]=$result;
                else {
                    $removed[]=$metadata['edits'][$i]['renderedFileId'];
                    $metadata['edits'][$i]=$result;
                }
            } elseif (in_array($action,[
                'folder.delete','video.delete','edit.delete'
            ],true)) {
                $folderIds=[];
                $videoIds=[];
                $editIds=[];
                if ($action==='folder.delete') {
                    indexOf($metadata['folders'],$entityId);
                    $folderIds=[$entityId];
                    do {
                        $count=count($folderIds);
                        foreach ($metadata['folders'] as $folder)
                            if (in_array($folder['parentFolderId'],$folderIds,true) &&
                                !in_array($folder['id'],$folderIds,true))
                                $folderIds[]=$folder['id'];
                    } while ($count!==count($folderIds));
                    foreach ($metadata['videos'] as $video)
                        if (in_array($video['folderId'],$folderIds,true))
                            $videoIds[]=$video['id'];
                } elseif ($action==='video.delete') {
                    indexOf($metadata['videos'],$entityId);
                    $videoIds=[$entityId];
                } else {
                    indexOf($metadata['edits'],$entityId);
                    $editIds=[$entityId];
                }
                foreach ($metadata['videos'] as $video)
                    if (in_array($video['id'],$videoIds,true))
                        $removed[]=$video['originalFileId'];
                foreach ($metadata['edits'] as $edit)
                    if (in_array($edit['videoId'],$videoIds,true) ||
                        in_array($edit['id'],$editIds,true)) {
                        $editIds[]=$edit['id'];
                        $removed[]=$edit['renderedFileId'];
                    }
                foreach ([
                    'folders'=>$folderIds,'videos'=>$videoIds,'edits'=>$editIds
                ] as $key=>$ids)
                    $metadata[$key]=array_values(array_filter(
                        $metadata[$key],fn($row)=>!in_array($row['id'],$ids,true)
                    ));
            } else {
                fail('NOT_FOUND','操作がありません。',404);
            }

            $deletions=readJson('deletion-retry.json');
            foreach ($metadata['files'] as $file)
                if (in_array($file['id'],$removed,true)) $deletions[]=$file;
            $metadata['files']=array_values(array_filter(
                $metadata['files'],fn($file)=>!in_array($file['id'],$removed,true)
            ));

            if ($incoming &&
                !rename($incoming['temp'],filePath($incoming['file'])))
                fail('SAVE_FAILED','動画を確定できません。',500);

            $requests[$requestId]=['hash'=>$hash,'data'=>$result];

            /*
             * transaction.json確定後はコミット済みとして復旧する。
             * 応答が途切れても同じrequestIdで結果を取得できる。
             */
            atomicWrite('transaction.json',[
                'metadata'=>$metadata,'requests'=>$requests,'deletions'=>$deletions
            ]);
            recover();
        } catch (Throwable $error) {
            if ($incoming && !is_file(ROOT.'/transaction.json')) {
                if (is_file($incoming['temp'])) @unlink($incoming['temp']);
                $path=filePath($incoming['file']);
                if (is_file($path)) @unlink($path);
            }
            throw $error;
        }

        answer($result,$requestId);
    }
} catch (Throwable $error) {
    if (is_resource($lock)) {
        flock($lock,LOCK_UN);
        fclose($lock);
    }
    error_log((string)$error);
    $status=$error instanceof AppError?$error->getCode():500;
    $message=$error instanceof AppError?$error->getMessage():
        'サーバ処理に失敗しました。PHPログを確認してください。';
    http_response_code($status);
    if (isset($_GET['api'])) {
        header('Content-Type: application/json; charset=utf-8');
        echo jsonText([
            'ok'=>false,'requestId'=>$requestId,
            'error'=>[
                'code'=>$error instanceof AppError?$error->kind:'SERVER_ERROR',
                'message'=>$message
            ]
        ]);
    } else {
        header('Content-Type: text/html; charset=utf-8');
        echo '<h1>起動エラー</h1><p>'.
            htmlspecialchars($message,ENT_QUOTES,'UTF-8').'</p>';
    }
    exit;
}

async function transmit(packet){
    const options={cache:'no-store'};
    if(packet.request){
        options.method='POST';
        if(packet.file){
            const body=new FormData();
            body.append('metadata',JSON.stringify(packet.request));
            body.append('file',packet.file,'video.mp4');
            options.body=body;
        }else{
            options.headers={'Content-Type':'application/json'};
            options.body=JSON.stringify(packet.request);
        }
    }
    let result;
    for(let attempt=0;attempt<3;attempt++){
        try{
            const response=await fetch(apiURL(packet.action,packet.id),options);
            const text=await response.text();
            try{result=JSON.parse(text)}
            catch{throw Error('サーバ応答を確認できません。')}
            if(!response.ok||!result.ok){
                const error=Error(result.error?.message||'要求に失敗しました。');
                error.confirmed=true;
                throw error;
            }
            if(state.pending===packet)state.pending=null;
            return result.data;
        }catch(error){
            if(error.confirmed){
                if(state.pending===packet)state.pending=null;
                throw error;
            }
            if(attempt===2)
                throw Error('通信結果を確認できません。同じ要求を再試行してください。');
            await new Promise(resolve=>setTimeout(resolve,500*(attempt+1)));
        }
    }
}

function mediaWait(events,test,action,timeout=30000){
    return new Promise((resolve,reject)=>{
        const clean=()=>{

            video.removeEventListener('error',error);

        const timer=setTimeout(()=>{
            clean();reject(Error('動画処理がタイムアウトしました。'));

        video.addEventListener('error',error);

    });
}

function pathFor(e){
    const path=new Path2D();
    if(e.shape==='ellipse')
        path.ellipse(e.x+e.w/2,e.y+e.h/2,e.w/2,e.h/2,0,0,Math.PI*2);
    else if(e.shape==='round')
        path.roundRect(e.x,e.y,e.w,e.h,Math.min(14,e.w/2,e.h/2));
    else path.rect(e.x,e.y,e.w,e.h);
    return path;
}
function stroke(c,e,path){
    c.lineWidth=e.width;c.strokeStyle=e.stroke;
    c.setLineDash(dashes[e.dash]);c.lineJoin='round';
    if(e.stroke!=='transparent'&&e.width>0)c.stroke(path);
}
function endpoint(e,p){
    if(e.shape==='ellipse'){
        const rx=e.w/2,ry=e.h/2;
        const a=Math.cos(p.angle),b=Math.sin(p.angle);
        const length=Math.hypot(a/rx,b/ry);
        return {x:e.x+rx+rx*a,y:e.y+ry+ry*b,nx:a/rx/length,ny:b/ry/length};
    }
    const r=p.ratio;
    return ({
        left:{x:e.x,y:e.y+e.h*r,nx:-1,ny:0},
        right:{x:e.x+e.w,y:e.y+e.h*r,nx:1,ny:0},
        top:{x:e.x+e.w*r,y:e.y,nx:0,ny:-1},
        bottom:{x:e.x+e.w*r,y:e.y+e.h,nx:0,ny:1}
    })[p.side];
}
function attach(e,p){
    if(e.shape==='ellipse')
        return {angle:Math.atan2((p.y-e.y-e.h/2)/(e.h/2),(p.x-e.x-e.w/2)/(e.w/2))};
    const choices=[
        {side:'left',distance:Math.abs(p.x-e.x),ratio:(p.y-e.y)/e.h},
        {side:'right',distance:Math.abs(p.x-e.x-e.w),ratio:(p.y-e.y)/e.h},
        {side:'top',distance:Math.abs(p.y-e.y),ratio:(p.x-e.x)/e.w},
        {side:'bottom',distance:Math.abs(p.y-e.y-e.h),ratio:(p.x-e.x)/e.w}
    ].sort((a,b)=>a.distance-b.distance);
    return {side:choices[0].side,ratio:Math.max(0,Math.min(1,choices[0].ratio))};
}
function paint(c,time,base=false){
    c.save();
    try{
        c.setTransform(1,0,0,1,0,0);
        c.clearRect(0,0,c.canvas.width,c.canvas.height);
        c.scale(c.canvas.width/state.data.sourceWidth,c.canvas.height/state.data.sourceHeight);
        if(base)c.drawImage(video,0,0,state.data.sourceWidth,state.data.sourceHeight);
        const elements=state.data.elements;
        for(const type of ['C','B','A']){
            for(const e of elements){
                if(e.type!==type||time<e.start||time>=e.end)continue;
                c.save();
                const path=pathFor(e);
                if(e.background!=='transparent'){c.fillStyle=e.background;c.fill(path)}
                if(type==='C'){
                    c.save();c.clip(path);
                    const sw=e.w/e.zoom,sh=e.h/e.zoom;
                    c.drawImage(video,e.x+(e.w-sw)/2,e.y+(e.h-sh)/2,sw,sh,e.x,e.y,e.w,e.h);
                    c.restore();
                }
                stroke(c,e,path);
                if(type==='A'){
                    c.clip(path);c.setLineDash([]);c.fillStyle=e.color;
                    c.font=e.size+'px '+(e.font==='mincho'?'serif':'sans-serif');
                    c.textBaseline='top';
                    let y=e.y+5;
                    for(const paragraph of e.text.split('\n')){
                        let line='';
                        for(const character of Array.from(paragraph)){
                            if(line&&c.measureText(line+character).width>e.w-10){
                                c.fillText(line,e.x+5,y);y+=e.size*1.2;line='';
                            }
                            line+=character;
                        }
                        c.fillText(line,e.x+5,y);y+=e.size*1.2;
                    }
                }
                c.restore();
            }
        }
        const byId=new Map(elements.map(e=>[e.id,e]));
        for(const e of elements){
            if(e.type!=='E')continue;
            const owner=byId.get(e.ownerId),comment=byId.get(e.commentId);
            if(!owner||!comment||time<owner.start||time>=owner.end)continue;
            const a=endpoint(owner,e.from),b=endpoint(comment,e.to);
            const distance=Math.hypot(b.x-a.x,b.y-a.y);
            if(distance<.001)continue;
            const path=new Path2D();path.moveTo(a.x,a.y);
            let before=a;
            if(e.line==='orthogonal'){
                const offset=Math.min(24,distance/3);
                const p={x:a.x+a.nx*offset,y:a.y+a.ny*offset};
                const q={x:b.x+b.nx*offset,y:b.y+b.ny*offset};
                path.lineTo(p.x,p.y);
                if(Math.abs(a.nx)>=Math.abs(a.ny))path.lineTo(q.x,p.y);
                else path.lineTo(p.x,q.y);
                path.lineTo(q.x,q.y);path.lineTo(b.x,b.y);before=q;
            }else if(e.line==='wave'){
                for(let i=1;i<=48;i++){
                    const t=i/48,w=Math.sin(t*Math.PI*2)*Math.min(16,distance/8);
                    const p={x:a.x+(b.x-a.x)*t-(b.y-a.y)/distance*w,
                        y:a.y+(b.y-a.y)*t+(b.x-a.x)/distance*w};
                    path.lineTo(p.x,p.y);if(i===47)before=p;
                }
            }else path.lineTo(b.x,b.y);
            c.save();stroke(c,e,path);c.setLineDash([]);c.fillStyle=e.stroke;
            if(e.stroke!=='transparent'&&e.terminal!=='none'){
                c.beginPath();const size=Math.max(6,e.width*3);
                if(e.terminal==='circle')c.arc(b.x,b.y,size/2,0,Math.PI*2);
                else{
                    const angle=Math.atan2(b.y-before.y,b.x-before.x);
                    c.moveTo(b.x,b.y);
                    c.lineTo(b.x-size*Math.cos(angle-.52),b.y-size*Math.sin(angle-.52));
                    c.lineTo(b.x-size*Math.cos(angle+.52),b.y-size*Math.sin(angle+.52));
                    c.closePath();
                }
                c.fill();
            }
            c.restore();
        }
    }finally{c.restore()}
}
function draw(){
    if(!state.data){context.clearRect(0,0,canvas.width,canvas.height);return}
    if(state.editing||state.exporting)paint(context,video.currentTime,false);
    else context.clearRect(0,0,canvas.width,canvas.height);
    if(!state.editing||state.exporting)return;
    for(const e of state.data.elements){
        if(!state.selected.has(e.id))continue;
        context.save();context.setLineDash([4,4]);context.strokeStyle='#228be6';
        if(['A','B','C'].includes(e.type)){
            context.strokeRect(e.x,e.y,e.w,e.h);
            context.fillStyle='#228be6';context.fillRect(e.x+e.w-6,e.y+e.h-6,12,12);
        }else if(e.type==='E'){
            const owner=state.data.elements.find(x=>x.id===e.ownerId);
            const comment=state.data.elements.find(x=>x.id===e.commentId);
            if(owner&&comment){
                for(const [target,p,color] of [[owner,e.from,'#00a060'],[comment,e.to,'#e88000']]){
                    const point=endpoint(target,p);
                    context.fillStyle=color;context.beginPath();
                    context.arc(point.x,point.y,7,0,Math.PI*2);context.fill();
                }
            }
        }
        context.restore();
    }
}

function skipRanges(){
    const ranges=state.data.elements.filter(e=>e.type==='D')
        .map(e=>({start:e.start,end:e.end})).sort((a,b)=>a.start-b.start);
    const merged=[];
    for(const e of ranges){
        const last=merged[merged.length-1];
        if(last&&e.start<=last.end)last.end=Math.max(last.end,e.end);
        else merged.push(e);
    }
    return merged;
}
function validateTimes(){
    for(const e of state.data.elements){
        if(e.type==='E')continue;
        if(e.type==='F'){
            if(!Number.isFinite(e.time)||e.time<0||e.time>=state.duration)
                throw Error('PDF位置が動画の範囲外です。');
        }else if(!Number.isFinite(e.start)||!Number.isFinite(e.end)||
            e.start<0||e.start>=e.end||e.end>state.duration)
            throw Error('表示区間が動画の範囲外です。');
    }
}

async function recordScreen(){
    if(!discard())return;
    const type=mp4Type();
    if(!navigator.mediaDevices?.getDisplayMedia)
        throw Error('HTTPSまたはlocalhostで開いてください。');
    const stream=await navigator.mediaDevices.getDisplayMedia({video:true,audio:true});
    try{
        const recorder=new MediaRecorder(stream,{mimeType:type});
        const chunks=[];reset();state.recording=recorder;
        recorder.ondataavailable=e=>{if(e.data.size)chunks.push(e.data)};
        recorder.onerror=()=>message('録画処理に失敗しました。');

            stream.getTracks().forEach(t=>t.stop());

                '録画 '+new Date().toLocaleString()));
        };
        stream.getVideoTracks()[0].onended=()=>{
            if(recorder.state!=='inactive')recorder.stop();
        };
        recorder.start(250);message('録画中です。録画停止で保存します。');
    }catch(error){
        stream.getTracks().forEach(t=>t.stop());state.recording=null;throw error;

async function renderMovie(){
    validateTimes();
    const type=mp4Type(),skips=skipRanges();
    const segments=[];let cursor=0;
    for(const skip of skips){
        if(skip.start>cursor)segments.push({start:cursor,end:skip.start});
        cursor=skip.end;
    }
    if(cursor<state.duration)segments.push({start:cursor,end:state.duration});
    if(!segments.length)throw Error('動画全体がスキップされています。');

    const output=document.createElement('canvas');
    output.width=state.data.sourceWidth;output.height=state.data.sourceHeight;
    const c=output.getContext('2d');
    const stream=output.captureStream(30);
    const graph=await audioGraph();
    for(const track of graph.destination.stream.getAudioTracks())
        stream.addTrack(track.clone());

    const recorder=new MediaRecorder(stream,{mimeType:type});
    const chunks=[];
    let stoppedResolve,stoppedReject;
    const stopped=new Promise((resolve,reject)=>{
        stoppedResolve=resolve;stoppedReject=reject;
    });
    recorder.ondataavailable=e=>{if(e.data.size)chunks.push(e.data)};
    recorder.onstop=()=>stoppedResolve(new Blob(chunks,{type:'video/mp4'}));
    recorder.onerror=()=>stoppedReject(Error('MP4生成に失敗しました。'));

    state.exporting=true;video.pause();
    let animation;
    const render=()=>{
        paint(c,video.currentTime,true);
        animation=requestAnimationFrame(render);
    };
    try{
        await seekTo(segments[0].start);
        paint(c,video.currentTime,true);render();
        recorder.start(250);
        for(let i=0;i<segments.length;i++){
            const segment=segments[i];
            if(i>0){
                await seekTo(segment.start);
                paint(c,video.currentTime,true);
                const resumed=new Promise(resolve=>
                    recorder.addEventListener('resume',resolve,{once:true}));
                recorder.resume();await resumed;
            }
            await video.play();
            await new Promise((resolve,reject)=>{
                const deadline=performance.now()+
                    (segment.end-segment.start)*1000+30000;
                const check=()=>{
                    if(video.error){reject(Error('元動画の再生に失敗しました。'));return}
                    if(video.currentTime>=segment.end||video.ended){resolve();return}
                    if(performance.now()>deadline){reject(Error('動画生成がタイムアウトしました。'));return}
                    requestAnimationFrame(check);
                };
                check();
            });
            video.pause();
            if(i<segments.length-1){
                const paused=new Promise(resolve=>
                    recorder.addEventListener('pause',resolve,{once:true}));
                recorder.pause();await paused;
            }
            message('MP4生成中… '+(i+1)+' / '+segments.length);
        }
        recorder.stop();
        return await stopped;
    }finally{
        cancelAnimationFrame(animation);video.pause();
        if(recorder.state!=='inactive')recorder.stop();
        stream.getTracks().forEach(t=>t.stop());
        state.exporting=false;draw();
    }
}

    if(state.editing&&!state.exporting&&!video.paused&&!video.seeking&&state.data){
        const skip=skipRanges().find(e=>video.currentTime>=e.start&&video.currentTime<e.end);
        if(skip){

            else video.currentTime=skip.end;

    }

}
